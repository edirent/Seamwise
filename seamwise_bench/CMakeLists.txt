cmake_minimum_required(VERSION 3.20)
project(seamwise_bench LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SEAMWISE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(SEAMWISE_HOST_SIMD "Host-specific build (separate label required)" OFF)
option(SEAMWISE_INDEXED_REDUCTION "Sparse support-driven pivot lookup (exact implementation optimization)" ON)
option(SEAMWISE_FLAT_BP "Contiguous BP messages with the verified LDPC update order" ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND_CMAKE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH ${PYBIND_CMAKE})
find_package(pybind11 CONFIG REQUIRED)
add_library(seamwise_core STATIC cpp/src/decoder.cpp)
set_target_properties(seamwise_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(seamwise_core PUBLIC cpp/include vendor/ldpc/src_cpp vendor/ldpc/include/robin_map)
target_compile_options(seamwise_core PUBLIC -fno-fast-math -ffp-contract=off -Wall -Wextra -Wno-sign-compare -Wno-unused-parameter)
if(SEAMWISE_INDEXED_REDUCTION)
  target_compile_definitions(seamwise_core PUBLIC SEAMWISE_INDEXED_REDUCTION=1)
endif()
if(SEAMWISE_FLAT_BP)
  target_compile_definitions(seamwise_core PUBLIC SEAMWISE_FLAT_BP=1)
endif()
if(SEAMWISE_HOST_SIMD)
  target_compile_options(seamwise_core PUBLIC -march=native)
endif()
if(SEAMWISE_SANITIZE)
  target_compile_options(seamwise_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(seamwise_core PUBLIC -fsanitize=address,undefined)
endif()
pybind11_add_module(_native NO_EXTRAS cpp/src/bindings.cpp)
target_link_libraries(_native PRIVATE seamwise_core)
add_executable(native_runner cpp/bench/native_runner.cpp)
target_link_libraries(native_runner PRIVATE seamwise_core)
add_executable(native_tests tests/native_tests.cpp)
target_link_libraries(native_tests PRIVATE seamwise_core)
enable_testing()
add_test(NAME native_algebra_graph COMMAND native_tests)
find_package(OpenMP REQUIRED)
pybind11_add_module(_gari NO_EXTRAS cpp/src/gari_bindings.cpp vendor/gari/my_decoders/hbp_decoder_v2.c)
target_compile_options(_gari PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(_gari PRIVATE OpenMP::OpenMP_C)
