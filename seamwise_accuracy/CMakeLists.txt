cmake_minimum_required(VERSION 3.20)
project(seamwise_accuracy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND_CMAKE OUTPUT_STRIP_TRAILING_WHITESPACE)
list(APPEND CMAKE_PREFIX_PATH ${PYBIND_CMAKE})
find_package(pybind11 CONFIG REQUIRED)
option(ACCURACY_SANITIZE "Sanitizer correctness build" OFF)
add_library(accuracy_core STATIC cpp/src/decoder.cpp)
set_target_properties(accuracy_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(accuracy_core PUBLIC cpp/include ../seamwise_bench/vendor/ldpc/src_cpp ../seamwise_bench/vendor/ldpc/include/robin_map)
target_compile_definitions(accuracy_core PUBLIC SEAMWISE_INDEXED_REDUCTION=1 SEAMWISE_FLAT_BP=1)
target_compile_options(accuracy_core PUBLIC -fno-fast-math -ffp-contract=off -Wall -Wextra -Wno-sign-compare -Wno-unused-parameter)
if(ACCURACY_SANITIZE)
 target_compile_options(accuracy_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(accuracy_core PUBLIC -fsanitize=address,undefined)
endif()
pybind11_add_module(_accuracy_native NO_EXTRAS cpp/src/bindings.cpp)
target_link_libraries(_accuracy_native PRIVATE accuracy_core)
add_executable(accuracy_tests tests/native_tests.cpp)
target_link_libraries(accuracy_tests PRIVATE accuracy_core)
